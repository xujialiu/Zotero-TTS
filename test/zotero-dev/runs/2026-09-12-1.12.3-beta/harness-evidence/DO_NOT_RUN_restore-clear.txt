  ];

    const entry=snap.prefs?.[name];
    if(!entry){restored.push({name,ok:false,reason:'not captured'});return}
    try{
      if(entry.user){
        if(entry.type===Services.prefs.PREF_STRING)Services.prefs.setStringPref(name,String(entry.value??''));
        else if(entry.type===Services.prefs.PREF_BOOL)Services.prefs.setBoolPref(name,!!entry.value);
        else if(entry.type===Services.prefs.PREF_INT)Services.prefs.setIntPref(name,Number(entry.value));
        else throw new Error('unsupported pref type '+entry.type);
      }else if(Services.prefs.prefHasUserValue(name))Services.prefs.clearUserPref(name);
      restored.push({name,ok:true,user:Services.prefs.prefHasUserValue(name),type:Services.prefs.getPrefType(name)});
    }catch(e){restored.push({name,ok:false,error:String(e)})}
  };
  for(const name of order)restore(name);

  const verify={};
  for(const name of order){

    let equal=!expected||user===expected.user;
    if(equal&&expected?.user){
